  <script>
    let formSubmitted = false;

    // 智能表單邏輯：出席狀態改變時切換顯示項目
    function handleAttendanceChange(selectedValue) {
      const conditionalFields = document.getElementById('conditional-fields');
      const countSelect = document.getElementById('entry_322241103');
      const companionInput = document.getElementById('entry_2046862057');
      const invitationSelect = document.getElementById('entry_135469609');

      if (selectedValue === '絕對出席，參與你們生命中重要的時刻') {
        conditionalFields.classList.remove('hidden');
        conditionalFields.classList.add('animate-fade-in');
        countSelect.setAttribute('required', 'required');
      } else {
        conditionalFields.classList.add('hidden');
        conditionalFields.classList.remove('animate-fade-in');
        countSelect.removeAttribute('required');
        
        // 清空隱藏項目的輸入值
        countSelect.value = '';
        companionInput.value = '';
        invitationSelect.value = '';
        handleInvitationChange(''); // 同步隱藏並清空地址
      }
    }

    // 智能喜帖邏輯：根據選擇是否展開寄送地址
    function handleInvitationChange(selectedValue) {
      const addressField = document.getElementById('address-field');
      const addressInput = document.getElementById('entry_1856739771');

      if (selectedValue === '我想要紙本喜帖留作紀念' || selectedValue === '小朋友才做選擇我兩個都想要') {
        addressField.classList.remove('hidden');
        addressField.classList.add('animate-fade-in');
        addressInput.setAttribute('required', 'required');
      } else {
        addressField.classList.add('hidden');
        addressField.classList.remove('animate-fade-in');
        addressInput.removeAttribute('required');
        addressInput.value = '';
      }
    }

    // 表單提交觸發標記
    function onFormSubmit() {
      formSubmitted = true;
      const btn = document.getElementById('submit-button');
      btn.disabled = true;
      btn.innerHTML = `<i class="fa-solid fa-circle-notch fa-spin text-sm"></i><span>傳送中...</span>`;
    }

    // Iframe 載入完成時處理成功響應
    function handleIframeLoad() {
      if (formSubmitted) {
        showSuccessModal();
        
        // 重設表單狀態
        const form = document.getElementById('wedding-rsvp-form');
        form.reset();
        
        // 重置按鈕與條件欄位
        const btn = document.getElementById('submit-button');
        btn.disabled = false;
        btn.innerHTML = `<i class="fa-regular fa-paper-plane text-sm"></i><span>確認送出</span>`;
        
        handleAttendanceChange('');
        formSubmitted = false;
      }
    }

    // 顯示自訂成功 Modal
    function showSuccessModal() {
      const modal = document.getElementById('success-modal');
      const content = modal.querySelector('.id-modal-content');
      
      modal.classList.remove('opacity-0', 'pointer-events-none');
      content.classList.remove('scale-95');
      content.classList.add('scale-100');
    }

    // 關閉 Modal
    closeSuccessModal = function() {
      const modal = document.getElementById('success-modal');
      const content = modal.querySelector('.id-modal-content');
      
      content.classList.remove('scale-100');
      content.classList.add('scale-95');
      modal.classList.add('opacity-0', 'pointer-events-none');
    }

    // 頁面載入完成初始化
    window.onload = function() {
      const iframe = document.getElementById('hidden_iframe');
      if (iframe) {
        iframe.addEventListener('load', handleIframeLoad);
      }
    };
  </script>
